\newcommand{\MegaTopic}[1]{%
    \refstepcounter{megatopicidx}%
    \clearpage%
    \phantomsection%
    \addcontentsline{toc}{megatopic}{\MakeUppercase{#1}}%
    \markboth{#1}{}%
    \thispagestyle{empty}%
    \begin{tikzpicture}[remember picture,overlay]
        \fill[blue!3]
            (current page.south west) rectangle (current page.north east);
        \draw[blue!18,line width=0.8pt,rounded corners=8pt]
            ([xshift=8mm,yshift=8mm]current page.south west)
            rectangle
            ([xshift=-8mm,yshift=-8mm]current page.north east);
    \end{tikzpicture}%
    \null\vspace*{\fill}%
    \begin{center}
        {\Huge\bfseries\MakeUppercase{#1}\par}%
        \vspace{13mm}
        \MegaTopicOrnament
    \end{center}%
    \vspace*{\fill}%
    \clearpage%
}

% Строка оглавления без \leftskip — иначе отступ «накапливается» при переносе страницы.
\newcommand{\@tocline}[4]{%
    \par\noindent
    \hb@xt@\linewidth{%
        \hspace*{#1}%
        \begingroup
            \@tempdima=#2\relax
            #3%
        \endgroup
        \nobreak\@tocdotfill
        \nobreak\hb@xt@\@pnumwidth{\hss #4}%
    }%
    \par
}

\newcommand*\l@megatopic[2]{%
    \ifnum \c@tocdepth >-3\relax
        \addpenalty{-\@highpenalty}%
        \addvspace{1.05em \@plus 2pt}%
        \par\noindent
        \hb@xt@\linewidth{%
            \@tocrulefill
            \@tocflourishleft\hspace{0.55em}%
            {\LARGE\bfseries\textcolor{blue!55!black}{#1}}%
            \hspace{0.55em}\@tocflourishright
            \@tocrulefill
        }%
        \par
        \nobreak
        \addvspace{0.35em}%
    \fi
}

\renewcommand*\l@part[2]{%
    \ifnum \c@tocdepth >-2\relax
        \addpenalty{-\@highpenalty}%
        \addvspace{0.85em \@plus 1pt}%
        \par\noindent
        \hb@xt@\linewidth{%
            \@tocrulefill
            \hspace{0.75em}%
            {\large\bfseries\textcolor{blue!55!black}{#1}}%
            \hspace{0.75em}%
            \@tocrulefill
        }%
        \par
        \addvspace{0.2em}%
    \fi
}

\renewcommand*\l@chapter[2]{%
    \ifnum \c@tocdepth >\m@ne

    \fi
}
