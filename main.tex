\newcommand{\Semester}[1]{%
    \clearpage%
    \phantomsection%
    \setcounter{part}{#1}%
    \addcontentsline{toc}{part}{\Roman{part} семестр}%
    \markboth{}{}%
    \thispagestyle{empty}%
    \vspace*{\fill}%
    \begin{center}
        {\Huge\bfseries \Roman{part} семестр}
    \end{center}
    \vspace*{\fill}%
    \clearpage%
    \setcounter{chapter}{0}%
}
